def run(label, argv):
    out, err = io.StringIO(), io.StringIO()
    started = time.monotonic()
    try:
        with contextlib.redirect_stdout(out), contextlib.redirect_stderr(err):
            code = cli.main(argv)
        result = f"exit {code}"
    except SystemExit as stop:
        result = f"SystemExit {stop.code}"
    except BaseException as error:  # noqa: BLE001
        result = f"TRACEBACK {type(error).__name__}: {str(error)[:90]}"
    text = (out.getvalue() + err.getvalue()).strip().replace("\n", " / ")[:150]
    print(f"{label:34s} -> {result:14s} [{time.monotonic() - started:5.1f}s] {text}")

def body(atoms, scale=1.0, sep=" "):
    return [f"{s}{sep}{x * scale:.6f}{sep}{y * scale:.6f}{sep}{z * scale:.6f}" for s, x, y, z in atoms]


def xyz(name, lines, raw=None):
    path = D / name
    if raw is not None:
        path.write_bytes(raw)
    else:
        path.write_text("\n".join(lines) + "\n")
    return path
