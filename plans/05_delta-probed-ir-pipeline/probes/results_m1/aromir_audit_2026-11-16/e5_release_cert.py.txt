def run(label, argv):
    out, err = io.StringIO(), io.StringIO()
    try:
        with contextlib.redirect_stdout(out), contextlib.redirect_stderr(err):
            code = cli.main(argv)
        result = f"exit {code}"
    except BaseException as error:  # noqa: BLE001
        result = f"TRACEBACK {type(error).__name__}: {str(error)[:110]}"
    text = err.getvalue().strip().replace("\n", " / ")[:170]
    print(f"{label:40s} -> {result} {text}")
    return out.getvalue()


def copy(name):
    folder = TMP / "rel" / name
    if folder.exists():
        shutil.rmtree(folder)
    shutil.copytree(release.bundled(), folder)
    return folder


def edit(path, change):
    record = json.loads(path.read_text())
    change(record)
    path.write_text(json.dumps(record))

print("--- certificate on corpus molecules")
r = rel()
rows = list(csv.DictReader((CORPUS.parent / "manifest.csv").open()))
by_smiles = {}
for row in rows:
    if (CORPUS / row["id"] / "hessian_b3lyp_analytic.npz").is_file() and (CORPUS / row["id"] / "dipole_b3lyp_cphf.npz").is_file():
        by_smiles.setdefault(row["smiles"], row)


def cert(row):
    g, h = load(row["id"])
    with np.load(CORPUS / row["id"] / "dipole_b3lyp_cphf.npz") as z:
        apt = np.array(z["apt"], dtype=float)
    low = LowLevel(0.0, h, apt, response.Refinement(0, 0, 0, 0), 0.0, 0.0)
    admitted = admission.Admission(row["smiles"], "?", g.n_atoms)
    return certificate.build(admitted, g, low, r.corrections(g, h), r)


for s in ("Oc1ccccc1", "Nc1ccc2ccccc2c1", "O=C(O)c1ccc2ccccc2c1", "Cc1ccc2ccccc2c1", "c1ccc2ccccc2c1", "c1ccc(-c2ccccc2)cc1"):
    if s not in by_smiles:
        print("  not available:", s)
        continue
    record = cert(by_smiles[s])
    bands = record["bands"]
    high = [(b["wavenumber_cm"], b["b3lyp_cm"], b["family"], b["licensed"]) for b in bands if b["b3lyp_cm"] > 3250]
    print(f"  {s}: families {dict(Counter(b['family'] for b in bands))}; bands above 3250: {high}")

sample = [row for row in by_smiles.values()]
rng = np.random.default_rng(3)
picked = [sample[k] for k in rng.choice(len(sample), size=60, replace=False)]
outside, n_bands, imaginary, failed, worst = 0, 0, 0, Counter(), 0.0
xh_licensed = 0
for row in picked:
    try:
        record = cert(row)
    except Exception as error:  # noqa: BLE001
        failed[type(error).__name__] += 1
        continue
    for b in record["bands"]:
        n_bands += 1
        low_edge, high_edge = b["seed_range_cm"]
        miss = max(low_edge - b["wavenumber_cm"], b["wavenumber_cm"] - high_edge)
        if miss > 0.05:
            outside += 1
            worst = max(worst, miss)
        imaginary += b["wavenumber_cm"] < 0
        xh_licensed += b["family"] == "CH-stretch" and b["b3lyp_cm"] > 3300
print(f"  60 random corpus molecules: failed {dict(failed)}; {n_bands} bands; printed position outside its own "
      f"'seeds' range' for {outside} bands (worst by {worst:.1f} cm-1); imaginary corrected bands {imaginary}; "
      f"bands above 3300 cm-1 called CH-stretch and licensed: {xh_licensed}")
